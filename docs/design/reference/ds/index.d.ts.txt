import type * as React from 'react';
export type Category = 'privacidad' | 'diy' | 'ia' | 'software' | 'linux';
export interface ButtonProps extends React.ButtonHTMLAttributes<HTMLButtonElement> {
  /** primary = ink fill (default) · secondary = outline · accent = mirla, once per view · text = underlined link-button */
  variant?: 'primary' | 'secondary' | 'accent' | 'text';
  size?: 'md' | 'sm';
  /** Adds a → that slides 4px on hover. */
  arrow?: boolean;
  /** Renders an <a> instead of a <button>. */
  href?: string;
}
export declare function Button(props: ButtonProps): React.ReactElement;
export interface CategoryTagProps { category: Category; /** Overrides the default label. */ children?: React.ReactNode; href?: string; className?: string }
export declare function CategoryTag(props: CategoryTagProps): React.ReactElement;
export interface NavLink { label: string; href: string; active?: boolean }
export interface NavBarProps { links?: NavLink[]; lang?: 'es' | 'en'; onLangChange?: (lang: 'es' | 'en') => void; homeHref?: string; className?: string }
export declare function NavBar(props: NavBarProps): React.ReactElement;
export interface PostCardProps {
  title: string; href?: string; excerpt?: string; category: Category;
  /** Display date, e.g. "23.09.2026"; dateTime is the ISO value. */
  date?: string; dateTime?: string; author?: string; readTime?: string;
  image?: string; imageAlt?: string;
  /** Large horizontal layout with HUD corner ticks. One per page. */
  featured?: boolean; eyebrow?: string; moreLabel?: string; className?: string;
}
export declare function PostCard(props: PostCardProps): React.ReactElement;
export interface NewsletterFormProps { title?: string; description?: string; eyebrow?: string; placeholder?: string; buttonLabel?: string; status?: 'idle' | 'success' | 'error'; message?: string; onSubmit?: (email: string) => void; id?: string; className?: string }
export declare function NewsletterForm(props: NewsletterFormProps): React.ReactElement;
export interface CodeBlockProps { /** Lines starting with "$ " get a mirla prompt, stripped on copy. */ code: string; lang?: string; filename?: string; showCopy?: boolean; className?: string }
export declare function CodeBlock(props: CodeBlockProps): React.ReactElement;
export interface CalloutProps { tone?: 'nota' | 'aviso' | 'peligro'; title?: string; children?: React.ReactNode; className?: string }
export declare function Callout(props: CalloutProps): React.ReactElement;
export interface LogoProps {
  /** Alto de la marca en px (el ancho es 1.31 × alto). */ size?: number;
  /** auto = sigue el tema (color en Día, blanco en Noche) · color · blanco · negro: las tres versiones oficiales. */
  variant?: 'auto' | 'color' | 'blanco' | 'negro';
  /** Añade "BogDev" compuesto en tipo junto a la marca. */ wordmark?: boolean; className?: string;
}
export declare function Logo(props: LogoProps): React.ReactElement;
declare global { interface Window { BogDev: { Button: typeof Button; CategoryTag: typeof CategoryTag; NavBar: typeof NavBar; PostCard: typeof PostCard; NewsletterForm: typeof NewsletterForm; CodeBlock: typeof CodeBlock; Callout: typeof Callout; Logo: typeof Logo } } }
